cmake_minimum_required(VERSION 3.16)
project(QuizIQ LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "QuizIQ's native interface requires Windows.")
endif()
add_executable(QuizIQ WIN32 src/main.cpp)
target_link_libraries(QuizIQ PRIVATE gdiplus shell32 ole32 user32 gdi32 uuid comdlg32)
if(MINGW)
  target_link_options(QuizIQ PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(QuizIQ PRIVATE /utf-8)
endif()
add_executable(engine_tests tests/engine_tests.cpp)
add_executable(literary_tests tests/literary_tests.cpp)
add_executable(view_tests tests/view_tests.cpp)
target_link_libraries(view_tests PRIVATE gdiplus shell32 ole32 user32 gdi32 uuid comdlg32)
if(MSVC)
  target_compile_options(view_tests PRIVATE /utf-8)
  target_compile_options(literary_tests PRIVATE /utf-8)
endif()
enable_testing()
add_test(NAME quiz_engine COMMAND engine_tests)
add_test(NAME native_views COMMAND view_tests)
add_test(NAME literary_notes COMMAND literary_tests)
